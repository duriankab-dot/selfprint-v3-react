// services/DecisionService.ts
export const createDecision = async (decision: Omit<Decision, 'id' | 'createdAt'>) => {
  const id = uuid();
  
  // สร้าง Follow-up schedule
  const followUps: FollowUp[] = [30, 90, 180, 365].map(days => ({
    id: uuid(),
    decisionId: id,
    days: days as FollowUp['days'],
    scheduledDate: addDays(new Date(), days),
    completed: false,
    notificationSent: false,
  }));
  
  // Save to Supabase
  await supabase.from('decisions').insert({
    id,
    ...decision,
    followUps,
    createdAt: new Date(),
  });
  
  // Schedule notifications
  for (const followUp of followUps) {
    await scheduleNotification({
      userId: decision.userId,
      type: 'decision_followup',
      scheduledFor: followUp.scheduledDate,
      data: { decisionId: id, days: followUp.days },
    });
  }
  
  return { id, followUps };
};

// services/sice/SICEBase.ts
export abstract class SICEBase {
  abstract id: number;
  abstract name: string;
  abstract description: string;
  
  abstract process(input: SICEInput): Promise<SICEOutput>;
  
  protected validateInput(input: any): boolean {
    return input && typeof input === 'object';
  }
  
  protected createResult(data: any, confidence: number): SICEOutput {
    return {
      data,
      confidence,
      source: this.name,
      timestamp: new Date(),
    };
  }
}

// services/sice/SICEOrchestrator.ts
export class SICEOrchestrator {
  private engines: Map<number, SICEBase> = new Map();
  
  constructor() {
    this.registerEngines();
  }
  
  private registerEngines() {
    this.engines.set(1, new PersonalContextBuilder());
    this.engines.set(2, new PatternDetector());
    // ... ทั้ง 12 ตัว
  }
  
  async orchestrate(input: UserInput): Promise<OrchestratorResult> {
    const results: SICEOutput[] = [];
    
    // Step 1: Run all engines in parallel
